        GameObject root;
        [UnityTearDown] public IEnumerator Cleanup(){if(root)UnityEngine.Object.Destroy(root);yield return null;}

        [UnityTest] public IEnumerator SpecialActivationPrecedesFragmentsAndInputUnlocksAfterSettle()
        {
            var s=Open();yield return null;
            foreach(int fixture in new[]{0,1,10})
            {
                s.Reset(fixture,GameplayRuleVersion.PrototypeR3);yield return null;
                var order=new List<GameplayMotionKind>();s.Board.MotionEventPublished+=e=>order.Add(e.Kind);
                string prefix=fixture==0?"LineR9":fixture==1?"BombR9":"SparkR9";
                MechanicsSandboxPlayModeTests.Capture(root,prefix+"_Before",576,1280);
                Assert.That(s.Board.TapCell(3,3),Is.True);Assert.That(s.Board.HasActiveCommand,Is.True);
                for(int frame=0;frame<10;frame++)
                {
                    yield return new WaitForSecondsRealtime(.03f);
                    MechanicsSandboxPlayModeTests.Capture(root,prefix+"_"+frame.ToString("D2"),576,1280);
                }
                float end=Time.realtimeSinceStartup+20;while(s.Board.HasActiveCommand&&Time.realtimeSinceStartup<end)yield return null;
                Assert.That(order.IndexOf(GameplayMotionKind.SpecialActivationStarted),Is.GreaterThanOrEqualTo(0));
                Assert.That(order.IndexOf(GameplayMotionKind.MatchWaveStarted),Is.GreaterThan(order.IndexOf(GameplayMotionKind.SpecialActivationStarted)));
                Assert.That(s.Board.HasActiveCommand,Is.False);Assert.That(s.Board.CanAcceptPointerInput,Is.True);
                Assert.That(s.Board.BoardStateHash,Is.EqualTo(s.Board.LastMechanicsCommand.Resolve().FinalSnapshot.StateHash));
                yield return new WaitForSecondsRealtime(.7f);
                Assert.That(s.Board.GetComponentInChildren<PrototypeImpactView>().ActiveCount,Is.Zero);
            }
        }

        [UnityTest] public IEnumerator GemFragmentsUseApprovedTexturesAndRecoverWithoutLeaking()
        {
            var s=Open();yield return null;
            var views=s.Board.GetComponentsInChildren<GemShatterView>();Assert.That(views.Length,Is.EqualTo(6));
            var cells=new List<BoardClearedCellInstruction>();
            for(int i=0;i<60;i++)cells.Add(new BoardClearedCellInstruction(new BoardPosition(i%8,i/8),new BoardCellData(i%6)));
            var cue=new GameplayMotionEvent(1,GameplayMotionKind.MatchWaveStarted,new BoardPosition(3,3),new BoardPosition(3,3),1,cells.Count,SpecialGemKind.None,"",clearedCells:cells);
            foreach(var view in views)view.SendMessage("Motion",cue);
            yield return new WaitForSecondsRealtime(.08f);Canvas.ForceUpdateCanvases();
            for(int i=0;i<views.Length;i++)
            {
                Assert.That(views[i].mainTexture,Is.SameAs(ProductionGameplayArt.Gem(i).texture));
                Assert.That(views[i].ActivePieces,Is.GreaterThan(0).And.LessThanOrEqualTo(128));
                Assert.That(views[i].canvasRenderer.GetMesh().vertexCount,Is.GreaterThan(0).And.LessThanOrEqualTo(384));
                Assert.That(views[i].raycastTarget,Is.False);
            }
            var recover=new GameplayMotionEvent(2,GameplayMotionKind.CommandRecovered,new BoardPosition(0,0),new BoardPosition(0,0),0,0,SpecialGemKind.None,"");
            foreach(var view in views){view.SendMessage("Motion",recover);Assert.That(view.ActivePieces,Is.Zero);view.SendMessage("Motion",cue);}
            yield return new WaitForSecondsRealtime(.6f);
            foreach(var view in views)Assert.That(view.ActivePieces,Is.Zero);
        }
        [UnityTest] public IEnumerator ImpactMeshIsBoundedAndDisableClearsPresentation()
        {
            var s=Open();yield return null;var effects=s.Board.GetComponentInChildren<PrototypeImpactView>();
            var cue=new GameplayMotionEvent(1,GameplayMotionKind.SpecialActivationStarted,new BoardPosition(3,3),new BoardPosition(3,3),1,1,SpecialGemKind.Burst,"");
            for(int i=0;i<240;i++)effects.SendMessage("Motion",cue);
            Assert.That(effects.ActiveCount,Is.LessThanOrEqualTo(192));
            yield return new WaitForSecondsRealtime(.14f);
            Canvas.ForceUpdateCanvases();var mesh=effects.canvasRenderer.GetMesh();Assert.That(mesh,Is.Not.Null);Assert.That(mesh.vertexCount,Is.LessThan(65535));Assert.That(effects.raycastTarget,Is.False);
            effects.gameObject.SetActive(false);Assert.That(effects.ActiveCount,Is.Zero);
            effects.gameObject.SetActive(true);yield return null;Assert.That(effects.ActiveCount,Is.Zero);
        }
