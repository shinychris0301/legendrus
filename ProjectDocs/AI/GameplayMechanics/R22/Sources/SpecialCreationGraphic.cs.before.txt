using UnityEngine;
using UnityEngine.Sprites;
using UnityEngine.UI;
namespace LegendsOfRus
{
    // Continuous assembly of the actual idle sprite. No generated replacement silhouette.
    [RequireComponent(typeof(CanvasRenderer))]
    public sealed class SpecialCreationGraphic : MaskableGraphic
    {
        static readonly Vector2[] Edge={new Vector2(-.5f,-.5f),new Vector2(-.12f,-.5f),new Vector2(.25f,-.5f),new Vector2(.5f,-.5f),new Vector2(.5f,0),new Vector2(.5f,.5f),new Vector2(.12f,.5f),new Vector2(-.25f,.5f),new Vector2(-.5f,.5f),new Vector2(-.5f,0)};
        Sprite art;RectTransform target;bool vertical,settling;float born,settleStart;
        public const float GatherSeconds=.32f,SettleSeconds=.40f;
        public bool Active {get;private set;}
        public bool Settling=>Active&&settling;
        public bool FullyAssembled=>Settling&&Time.unscaledTime-born>=.24f;
        public Sprite Artwork=>art;
        public override Texture mainTexture=>art?art.texture:base.mainTexture;
        public void Begin(Sprite sprite,RectTransform anchor,bool column)
        {art=sprite;target=anchor;vertical=column;settling=false;born=Time.unscaledTime;Active=true;raycastTarget=false;SetMaterialDirty();SetVerticesDirty();}
        static float GatherProgress(float age)=>age<GatherSeconds?Mathf.SmoothStep(0,1,age/GatherSeconds)*.72f:.72f+.22f*(1-Mathf.Exp(-(age-GatherSeconds)*9));
        public void Settle(){settleStart=GatherProgress(Time.unscaledTime-born);settling=true;born=Time.unscaledTime;SetVerticesDirty();}
        public void Finish(){Active=false;SetVerticesDirty();}
        void Update(){if(Active){if(Time.unscaledTime-born>8)Finish();else SetVerticesDirty();}}
        protected override void OnDisable(){Finish();base.OnDisable();}
        protected override void OnPopulateMesh(VertexHelper h)
        {
            h.Clear();if(!Active||!art||!target)return;
            float age=Time.unscaledTime-born;
            float progress=settling?Mathf.Lerp(settleStart,1,Mathf.SmoothStep(0,1,age/.24f)):GatherProgress(age);
            float scale=settling?1+.085f*Mathf.Sin(Mathf.Clamp01(age/SettleSeconds)*Mathf.PI):1;
            Vector2 size=target.rect.size*1.08f;float aspect=art.rect.width/art.rect.height;
            if(size.x/size.y>aspect)size.x=size.y*aspect;else size.y=size.x/aspect;
            var uv=DataUtility.GetOuterUV(art);float unit=Mathf.Min(size.x,size.y);
            // Once joined, one quad gives a seam-free hold before the actual Image takes over.
            if(progress>=.9999f)
            {
                int n=h.currentVertCount;Add(h,new Vector2(-.5f,-.5f),Vector2.zero,0,size,scale,uv,1);Add(h,new Vector2(-.5f,.5f),Vector2.zero,0,size,scale,uv,1);Add(h,new Vector2(.5f,.5f),Vector2.zero,0,size,scale,uv,1);Add(h,new Vector2(.5f,-.5f),Vector2.zero,0,size,scale,uv,1);h.AddTriangle(n,n+1,n+2);h.AddTriangle(n,n+2,n+3);return;
            }
            for(int i=0;i<Edge.Length;i++)
            {
                Vector2 a=Edge[i],b=Edge[(i+1)%Edge.Length],center=(a+b)/3;
                float remaining=1-progress,angle=(i%2==0?1:-1)*remaining*.65f;
                Vector2 radial=center.normalized,side=new Vector2(-radial.y,radial.x);
                Vector2 offset=(radial*remaining*.52f+side*Mathf.Sin(remaining*Mathf.PI)*.13f)*unit;
                float alpha=Mathf.SmoothStep(0,1,Mathf.Clamp01(progress/.42f));
                int n=h.currentVertCount;Add(h,Vector2.zero,offset,angle,size,scale,uv,alpha);Add(h,a,offset,angle,size,scale,uv,alpha);Add(h,b,offset,angle,size,scale,uv,alpha);h.AddTriangle(n,n+1,n+2);
            }
        }
        void Add(VertexHelper h,Vector2 point,Vector2 offset,float angle,Vector2 size,float scale,Vector4 uv,float alpha)
        {
            Vector2 p=Vector2.Scale(point,size);float c=Mathf.Cos(angle),s=Mathf.Sin(angle);p=(new Vector2(p.x*c-p.y*s,p.x*s+p.y*c)+offset)*scale;
            if(vertical)p=new Vector2(-p.y,p.x);
            Vector3 world=target.TransformPoint(p);Vector3 local=rectTransform.InverseTransformPoint(world);
            h.AddVert(local,new Color(1,1,1,alpha),new Vector2(Mathf.Lerp(uv.x,uv.z,point.x+.5f),Mathf.Lerp(uv.y,uv.w,point.y+.5f)));
        }
    }
}
